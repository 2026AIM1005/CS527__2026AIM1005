% Vector FIR example.
% N is at address 0.
% Eight weights start at address 4.
% Input starts at address 36.
% Output starts at address 256.
% Temporary vector products use address 512.
%
% For each output, eight products are calculated in parallel.

x1 = [0]
x4 = x1 + 0
x24 = 0
x2 = 36
x3 = 255
x3 = x3 + 1
x15 = 128

.loop
x14 = x24 - x4
BEQ .finish

v0 = [x2]
v1 = [4]
v2 = v0 * v1
[x15] = v2

x16 = x15 + 0
x17 = x15 + 4
x18 = x15 + 8
x19 = x15 + 12
x20 = x15 + 16
x21 = x15 + 20
x22 = x15 + 24
x23 = x15 + 28

x5 = [x16]
x6 = [x17]
x7 = [x18]
x8 = [x19]
x9 = [x20]
x10 = [x21]
x11 = [x22]
x12 = [x23]
x13 = x5 + x6
x13 = x13 + x7
x13 = x13 + x8
x13 = x13 + x9
x13 = x13 + x10
x13 = x13 + x11
x13 = x13 + x12
[x3] = x13

x24 = x24 + 1
x2 = x2 + 4
x3 = x3 + 4
BAL .loop

.finish
